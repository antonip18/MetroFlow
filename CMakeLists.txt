cmake_minimum_required(VERSION 3.28)
project(MetroFlow VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(METROFLOW_USE_SYSTEM_SFML "Use an installed SFML 3 package" OFF)
option(METROFLOW_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
option(METROFLOW_RENDER_TESTS "Run render smoke tests (requires a display)" OFF)
include(CTest)
if(METROFLOW_USE_SYSTEM_SFML)
  find_package(SFML 3 REQUIRED COMPONENTS Graphics)
else()
  include(FetchContent)
  set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 0fa201c969e48ecc253581c5841ce73f44d42f49 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(SFML)
endif()
add_library(metroflow_core
  src/Passenger.cpp src/Station.cpp src/StationHandler.cpp src/Line.cpp
  src/LineHandler.cpp src/LineColor.cpp src/Curves.cpp src/WorldGeometry.cpp
  src/RandomShapeGenerator.cpp src/PassengerRouter.cpp src/Simulation.cpp
  src/TrackGeometry.cpp src/TrainGeometry.cpp)
target_include_directories(metroflow_core PUBLIC include)
target_link_libraries(metroflow_core PUBLIC SFML::Graphics)
add_library(metroflow_ui src/ShapeDrawer.cpp src/LineSelector.cpp src/DrawingHandler.cpp src/GameEventsHandler.cpp)
target_link_libraries(metroflow_ui PUBLIC metroflow_core)
add_executable(MetroFlow src/main.cpp)
target_link_libraries(MetroFlow PRIVATE metroflow_ui)
foreach(target metroflow_core metroflow_ui MetroFlow)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if(METROFLOW_SANITIZERS)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
add_custom_command(TARGET MetroFlow POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:MetroFlow>/assets")
if(BUILD_TESTING)
  add_executable(metroflow_tests tests/CoreTests.cpp)
  target_link_libraries(metroflow_tests PRIVATE metroflow_core)
  add_test(NAME core COMMAND metroflow_tests)
  add_executable(metroflow_track_tests tests/TrackTests.cpp)
  target_link_libraries(metroflow_track_tests PRIVATE metroflow_core)
  add_test(NAME track_layout COMMAND metroflow_track_tests)
  add_executable(metroflow_interaction_tests tests/InteractionTests.cpp)
  target_link_libraries(metroflow_interaction_tests PRIVATE metroflow_ui)
  target_compile_definitions(metroflow_interaction_tests PRIVATE METROFLOW_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_test(NAME interaction COMMAND metroflow_interaction_tests)
  add_executable(metroflow_render_tests tests/RenderTests.cpp)
  target_link_libraries(metroflow_render_tests PRIVATE metroflow_ui)
  target_compile_definitions(metroflow_render_tests PRIVATE METROFLOW_TEST_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  if(METROFLOW_RENDER_TESTS)
    add_test(NAME render COMMAND metroflow_render_tests "${CMAKE_CURRENT_BINARY_DIR}/screenshots")
  endif()
endif()
